
Create PROCEDURE [StandardNarrative].[MonthlyProductionDeterminedReportDetail]
    @From DATETIME,
    @To   DATETIME,
    @PersonId INT 
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        DISTINCT es.EpisodeId,
        ed.CycisID,
        (ed.LastName + ', ' + ed.FirstName) NAME,
        ed.OpenDate,
        es.Changed CompletedDate
    FROM StandardNarrative.EpisodeStatus es
        INNER JOIN StandardNarrative.Status s
            ON s.Id = es.StatusId
        INNER JOIN StandardNarrative.EpisodeAssignments ea
            ON ea.EpisodeId = es.EpisodeId
        INNER JOIN StandardNarrative.Users u
            ON u.Id = ea.UserId
        INNER JOIN dbo.People p
            ON p.Id = u.PersonId
        INNER JOIN StandardNarrative.EpisodeDemographics ed ON ed.EpisodeId  = es.EpisodeId
    WHERE
        s.Id = 3
        AND es.Changed >= @From
        AND es.Changed < DATEADD(DAY, 1, @To)
        AND p.Id = @PersonId 
    GROUP BY
        p.Id,
        ed.CycisID,
        ed.FirstName,
        ed.LastName,
        es.EpisodeId,
        ed.OpenDate,
        es.Changed
END;
GO
